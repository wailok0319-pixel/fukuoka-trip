    // 儲存至 localStorage 並更新狀態指示燈

    function saveState() {

        try {

            localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));

            const badge = document.getElementById("status-badge");

            badge.textContent = "💾 已儲存";

        } catch (e) {

            const badge = document.getElementById("status-badge");

            badge.textContent = "⚠️ 儲存失敗";

            badge.style.color = "#d9534f";

        }

    }


    // 從 localStorage 還原資料

    function loadState() {

        const saved = localStorage.getItem(STORAGE_KEY);

        if (saved) {

            try {

                const parsed = JSON.parse(saved);

                if (parsed.version === APP_VERSION) {

                    appState = parsed;

                }

            } catch (e) {

                console.error("還原資料失敗，使用預設值", e);

            }

        }

    }


    // 匯率計算邏輯

    function convertCurrency(source) {

        const jpyInput = document.getElementById("jpy-input");

        const hkdInput = document.getElementById("hkd-input");


        if (source === 'JPY') {

            const val = parseFloat(jpyInput.value);

            hkdInput.value = isNaN(val) ? "" : (val * exchangeRate).toFixed(1);

        } else {

            const val = parseFloat(hkdInput.value);

            jpyInput.value = isNaN(val) ? "" : Math.round(val / exchangeRate);

        }

    }

    function renderDateNav() {

            btn.onclick = () => {

                appState.selectedDate = dateKey;

                saveState();

                renderDateNav();

            };

            nav.appendChild(btn);

        });

    }
